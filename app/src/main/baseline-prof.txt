# Baseline Profile for Square
# Pre-compiles critical code ahead of time (AOT) to eliminate JIT pauses,
# reduce cold-start latency, and maintain 60-120 FPS during initial gestures.
#
# What to compile ahead of time, so the first fold of the bar is as smooth as
# the tenth.
#
# The bar was stuttering the first time it moved and settling after that, which
# is not the animation: it is the code being compiled while it runs. Forcing the
# whole app to compile ahead of time on the phone took the same cold gesture
# from 43% dropped frames and 85ms at the 90th percentile to 28% and 57 — the
# same numbers a warm run gives. These rules ship that head start.

# 1. Preload application and UI classes
Ldev/lelonio/square/**;

# 2. Jetpack Compose UI methods with complete signatures (Composer, bitmask, and return types)
# Explicit signature with parameters and types ensures the AOT compiler does not silently discard the profile.
HSPLdev/lelonio/square/ui/**;->*(**Landroidx/compose/runtime/Composer;*)V
HSPLdev/lelonio/square/ui/**;->*(**Landroidx/compose/runtime/Composer;I)V
HSPLdev/lelonio/square/ui/**;->*(**Landroidx/compose/runtime/Composer;II)V
HSPLdev/lelonio/square/ui/**;->*(Landroidx/compose/runtime/Composer;I)V
HSPLdev/lelonio/square/ui/**;->*(Landroidx/compose/runtime/Composer;II)V
HSPLdev/lelonio/square/ui/**;->*(**Landroidx/compose/runtime/Composer;*)Lkotlin/Unit;
HSPLdev/lelonio/square/ui/**;->*(**Landroidx/compose/runtime/Composer;*)**

# 3. Whole UI, playback, and data layers
HSPLdev/lelonio/square/ui/**;->*(**)**
HSPLdev/lelonio/square/playback/**;->*(**)**
HSPLdev/lelonio/square/data/**;->*(**)**
